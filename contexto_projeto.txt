// ============ ÁUDIO (Web Audio API) ============
// Todos os sons são gerados por síntese — não precisa de arquivos

function getAudioCtx() {
  if (!Estado.audioCtx) {
    Estado.audioCtx = new (window.AudioContext || window.webkitAudioContext)();
  }
  if (Estado.audioCtx.state === 'suspended') Estado.audioCtx.resume();
  return Estado.audioCtx;
}

function tocarTom(freq, dur, type = 'sine', volume = 0.15, delay = 0) {
  if (!Estado.somAtivado) return;
  try {
    const ctx = getAudioCtx();
    const osc = ctx.createOscillator();
    const gain = ctx.createGain();
    osc.type = type;
    osc.frequency.value = freq;
    osc.connect(gain);
    gain.connect(ctx.destination);
    const start = ctx.currentTime + delay;
    gain.gain.setValueAtTime(0, start);
    gain.gain.linearRampToValueAtTime(volume, start + 0.01);
    gain.gain.exponentialRampToValueAtTime(0.001, start + dur);
    osc.start(start);
    osc.stop(start + dur);
  } catch (e) { console.warn('[audio] Erro:', e); }
}

function somAcerto() {
  tocarTom(523.25, 0.15, 'sine', 0.18, 0);
  tocarTom(659.25, 0.15, 'sine', 0.18, 0.08);
  tocarTom(783.99, 0.20, 'sine', 0.18, 0.16);
  tocarTom(1046.5, 0.35, 'sine', 0.20, 0.24);
}

function somErro() {
  tocarTom(220, 0.15, 'sawtooth', 0.12, 0);
  tocarTom(185, 0.20, 'sawtooth', 0.12, 0.10);
  tocarTom(138.6, 0.35, 'sawtooth', 0.14, 0.22);
}

function somClique() {
  tocarTom(880, 0.06, 'square', 0.06, 0);
}

function somCombo(nivel) {
  const base = 523.25 * Math.pow(1.12, Math.min(nivel, 8));
  tocarTom(base, 0.1, 'triangle', 0.15, 0);
  tocarTom(base * 1.5, 0.15, 'triangle', 0.15, 0.08);
}

function somVitoria() {
  const notas = [523.25, 659.25, 783.99, 1046.5, 783.99, 1046.5, 1318.5];
  notas.forEach((f, i) => tocarTom(f, 0.25, 'sine', 0.18, i * 0.13));
  notas.forEach((f, i) => tocarTom(f * 0.5, 0.25, 'triangle', 0.08, i * 0.13));
}

function somDerrota() {
  const notas = [392, 349.23, 293.66, 220];
  notas.forEach((f, i) => tocarTom(f, 0.4, 'sine', 0.15, i * 0.22));
}

function explodirParticulas(x, y, qtd = 25, cores = null) {
  const palette = cores || ['#00498B', '#0677CC', '#00BDB9', '#FBAB30', '#96100E', '#D66C25'];
  for (let i = 0; i < qtd; i++) {
    const p = document.createElement('div');
    p.className = 'particle';
    const cor = palette[Math.floor(Math.random() * palette.length)];
    p.style.background = cor;
    p.style.boxShadow = `0 0 15px ${cor}, 0 0 30px ${cor}`;
    p.style.left = x + 'px';
    p.style.top = y + 'px';
    const angulo = Math.random() * Math.PI * 2;
    const dist = Math.random() * 220 + 80;
    p.style.setProperty('--tx', Math.cos(angulo) * dist + 'px');
    p.style.setProperty('--ty', Math.sin(angulo) * dist + 'px');
    const size = Math.random() * 10 + 6;
    p.style.width = size + 'px';
    p.style.height = size + 'px';
    p.style.animationDuration = (Math.random() * 0.5 + 0.7) + 's';
    document.body.appendChild(p);
    setTimeout(() => p.remove(), 1500);
  }
}

function burstConfetti(amount) {
  const colors = ['#00498B', '#0677CC', '#00BDB9', '#FBAB30', '#96100E', '#D66C25'];
  for (let i = 0; i < amount; i++) {
    const conf = document.createElement('div');
    conf.className = 'confetti';
    conf.style.left = Math.random() * 100 + 'vw';
    conf.style.background = colors[Math.floor(Math.random() * colors.length)];
    conf.style.animationDuration = (Math.random() * 2.5 + 2) + 's';
    conf.style.animationDelay = (Math.random() * 0.8) + 's';
    conf.style.animationName = 'fall';
    conf.style.animationTimingFunction = 'ease-in';
    conf.style.animationFillMode = 'forwards';
    conf.style.borderRadius = Math.random() > 0.5 ? '50%' : '2px';
    const size = Math.random() * 10 + 6;
    conf.style.width = conf.style.height = size + 'px';
    document.body.appendChild(conf);
    setTimeout(() => conf.remove(), 5000);
  }
}

function carregarJSONs() {
  return Promise.all([
    fetch('questoes.json').then(r => {
      if (!r.ok) throw new Error('Erro ao carregar questoes.json');
      return r.json();
    }),
    fetch('sobre.json').then(r => {
      if (!r.ok) throw new Error('Erro ao carregar sobre.json');
      return r.json();
    }).catch(() => null)
  ]);
}

function inicializar() {
  console.log('[DEBUG] Inicializando quiz...');

  // 1) Carrega config do localStorage
  carregarConfig();
  carregarSom();

  // 2) Configura eventos dos botões e modais
  configurarEventosModalConfig();
  configurarEventosModalSobre();
  configurarEventosQuiz();

  // 3) Carrega vozes
  speechSynthesis.onvoiceschanged = listarVozes;
  listarVozes();
  setTimeout(listarVozes, 200);

  // 4) Carrega JSONs e inicia o quiz
  carregarJSONs()
    .then(([questoes, sobre]) => {
      Estado.quizData = questoes;
      Estado.sobreData = sobre;

      if (DOM.loading) DOM.loading.classList.add('hidden');
      if (DOM.quizMain) DOM.quizMain.classList.remove('hidden');

      if (sobre) {
        renderizarSobre(sobre);
      } else if (DOM.aboutBody) {
        DOM.aboutBody.innerHTML =
          '<div class="about-descricao">Não foi possível carregar <b>sobre.json</b>.</div>';
      }

      loadQuestion();
    })
    .catch(err => {
      if (DOM.loading) {
        DOM.loading.innerHTML =
          `❌ Erro ao carregar <b>questoes.json</b>.<br><small>${err.message}</small>`;
      }
      console.error('[ERRO] Falha ao carregar JSONs:', err);
    });
}

// Aguarda o DOM estar pronto

  document.addEventListener('DOMContentLoaded', inicializar);
} else {
  inicializar();
}

// ============ MODAL DE CONFIGURAÇÕES ============

function abrirModal() {
  configTemp = { ...configAtiva };

  selecionarVozNoSelect(configTemp.voiceName, configTemp.voiceLang);

  if (DOM.rateRange) DOM.rateRange.value = configTemp.rate;
  if (DOM.pitchRange) DOM.pitchRange.value = configTemp.pitch;
  if (DOM.volumeRange) DOM.volumeRange.value = configTemp.volume;
  if (DOM.timerAtivoCheck) DOM.timerAtivoCheck.checked = configTemp.timerAtivo;
  if (DOM.tempoRange) DOM.tempoRange.value = configTemp.tempoPergunta;
  if (DOM.tempoFieldGroup) DOM.tempoFieldGroup.classList.toggle('disabled', !configTemp.timerAtivo);

  atualizarLabels();
  if (DOM.settingsModal) DOM.settingsModal.classList.add('show');
  somClique();
}

function fecharModal() {
  if (DOM.settingsModal) DOM.settingsModal.classList.remove('show');
  speechSynthesis.cancel();
}

function atualizarLabels() {
  if (DOM.rateValue && DOM.rateRange)
    DOM.rateValue.textContent = parseFloat(DOM.rateRange.value).toFixed(2) + 'x';
  if (DOM.pitchValue && DOM.pitchRange)
    DOM.pitchValue.textContent = parseFloat(DOM.pitchRange.value).toFixed(2);
  if (DOM.volumeValue && DOM.volumeRange)
    DOM.volumeValue.textContent = Math.round(DOM.volumeRange.value * 100) + '%';
  if (DOM.tempoValue && DOM.tempoRange)
    DOM.tempoValue.textContent = DOM.tempoRange.value + 's';
}

function configurarEventosModalConfig() {
  if (DOM.settingsBtn) DOM.settingsBtn.addEventListener('click', abrirModal);
  if (DOM.modalCloseBtn) DOM.modalCloseBtn.addEventListener('click', fecharModal);
  if (DOM.cancelBtn) DOM.cancelBtn.addEventListener('click', fecharModal);

  if (DOM.settingsModal) {
    DOM.settingsModal.addEventListener('click', e => {
      if (e.target === DOM.settingsModal) fecharModal();
    });
  }

  if (DOM.voiceSelect) {
    DOM.voiceSelect.addEventListener('change', () => {
      const voz = obterVozTemp();
      if (voz) {
        configTemp.voiceName = voz.name;
        configTemp.voiceLang = voz.lang;
      }
      somClique();
    });
  }

  if (DOM.rateRange) {
    DOM.rateRange.addEventListener('input', () => {
      configTemp.rate = parseFloat(DOM.rateRange.value);
      atualizarLabels();
    });
  }

  if (DOM.pitchRange) {
    DOM.pitchRange.addEventListener('input', () => {
      configTemp.pitch = parseFloat(DOM.pitchRange.value);
      atualizarLabels();
    });
  }

  if (DOM.volumeRange) {
    DOM.volumeRange.addEventListener('input', () => {
      configTemp.volume = parseFloat(DOM.volumeRange.value);
      atualizarLabels();
    });
  }

  if (DOM.timerAtivoCheck) {
    DOM.timerAtivoCheck.addEventListener('change', () => {
      configTemp.timerAtivo = DOM.timerAtivoCheck.checked;
      if (DOM.tempoFieldGroup) {
        DOM.tempoFieldGroup.classList.toggle('disabled', !configTemp.timerAtivo);
      }
      somClique();
    });
  }

  if (DOM.tempoRange) {
    DOM.tempoRange.addEventListener('input', () => {
      configTemp.tempoPergunta = parseInt(DOM.tempoRange.value, 10);
      atualizarLabels();
    });
  }

  if (DOM.testVoiceBtn) {
    DOM.testVoiceBtn.addEventListener('click', () => {
      const voz = obterVozTemp();
      if (voz) {
        configTemp.voiceName = voz.name;
        configTemp.voiceLang = voz.lang;
      }
      falar('Esta é uma demonstração da voz escolhida. Um, dois, três, testando.',
            { useTemp: true });
    });
  }

  if (DOM.resetConfigBtn) {
    DOM.resetConfigBtn.addEventListener('click', () => {
      configTemp = {
        voiceName: null, voiceLang: null,
        rate: 1, pitch: 1, volume: 1,
        timerAtivo: true, tempoPergunta: 30
      };
      selecionarVozNoSelect(null, null);
      if (DOM.rateRange) DOM.rateRange.value = 1;
      if (DOM.pitchRange) DOM.pitchRange.value = 1;
      if (DOM.volumeRange) DOM.volumeRange.value = 1;
      if (DOM.timerAtivoCheck) DOM.timerAtivoCheck.checked = true;
      if (DOM.tempoRange) DOM.tempoRange.value = 30;
      if (DOM.tempoFieldGroup) DOM.tempoFieldGroup.classList.remove('disabled');
      atualizarLabels();
      somClique();
    });
  }

  if (DOM.saveBtn) {
    DOM.saveBtn.addEventListener('click', () => {
      const voz = obterVozTemp();
      if (voz) {
        configTemp.voiceName = voz.name;
        configTemp.voiceLang = voz.lang;
      }
      configAtiva = { ...configTemp };
      salvarConfig();
      fecharModal();
      mostrarToast('✓ Configurações salvas!');
      somAcerto();

      if (!Estado.answered && Estado.quizData) iniciarTimer();
    });
  }
}

function iniciais(nome) {
  return nome.trim().split(/\s+/).slice(0, 2)
    .map(p => p[0] ? p[0].toUpperCase() : '').join('');
}

function criarLinks(pessoa) {
  const links = [];
  if (pessoa.linkedin && pessoa.linkedin.trim() !== '') {
    links.push(`<a class="about-link linkedin" href="${pessoa.linkedin}" target="_blank" rel="noopener" title="LinkedIn de ${pessoa.nome}">in</a>`);
  }
  if (pessoa.lattes && pessoa.lattes.trim() !== '') {
    links.push(`<a class="about-link lattes" href="${pessoa.lattes}" target="_blank" rel="noopener" title="Lattes de ${pessoa.nome}">CV</a>`);
  }
  return links.length ? `<div class="about-links">${links.join('')}</div>` : '';
}

function criarPessoa(pessoa, isProfessor = false) {
  const avatarClass = isProfessor ? 'about-avatar professor' : 'about-avatar';
  return `
    <div class="about-pessoa">
      <div class="about-pessoa-info">
        <div class="${avatarClass}">${iniciais(pessoa.nome)}</div>
        <div class="about-pessoa-nome">${pessoa.nome}</div>
      </div>
      ${criarLinks(pessoa)}
    </div>
  `;
}

function renderizarSobre(data) {
  const professorHTML = data.professor ? `
    <div class="about-secao">
      <div class="about-secao-titulo">${data.professor.titulo || 'Professor'}</div>
      ${criarPessoa(data.professor, true)}
    </div>` : '';

  const alunosHTML = (data.alunos && data.alunos.length) ? `
    <div class="about-secao">
      <div class="about-secao-titulo">Alunos</div>
      ${data.alunos.map(a => criarPessoa(a, false)).join('')}
    </div>` : '';

  if (DOM.aboutBody) {
    DOM.aboutBody.innerHTML = `
      <div>
        <div class="about-titulo">${data.titulo || 'Sobre o Projeto'}</div>
        <div class="about-descricao">${data.descricao || ''}</div>
      </div>
      ${professorHTML}
      ${alunosHTML}
      ${data.rodape ? `<div class="about-rodape">${data.rodape}</div>` : ''}
    `;
  }
}

function loadQuestion() {
  console.log('[DEBUG] loadQuestion — currentIndex:', Estado.currentIndex);

  Estado.answered = false;
  Estado.selectedIndex = null;

  if (!Estado.quizData || !Estado.quizData.questions[Estado.currentIndex]) {
    console.error('[ERRO] quizData inválido');
    return;
  }

  const q = Estado.quizData.questions[Estado.currentIndex];

  if (DOM.questionText) {
    DOM.questionText.textContent = q.question;
    DOM.questionText.style.animation = 'none';
    void DOM.questionText.offsetWidth;
    DOM.questionText.style.animation = '';
  }

  if (DOM.options) {
    DOM.options.innerHTML = '';

    q.options.forEach((opt, i) => {
      const btn = document.createElement('button');
      btn.className = 'option';
      btn.dataset.index = i;
      const spanText = document.createElement('span');
      spanText.textContent = opt;
      btn.appendChild(spanText);
      btn.addEventListener('click', () => selectOption(i, btn));
      DOM.options.appendChild(btn);
    });
  }

  // Reseta botões
  if (DOM.confirmBtn) {
    DOM.confirmBtn.style.display = 'none';
    DOM.confirmBtn.disabled = true;
  }
  if (DOM.nextBtn) {
    DOM.nextBtn.style.display = 'none';
  }

  if (DOM.progressInfo) {
    DOM.progressInfo.textContent =
      `Pergunta ${Estado.currentIndex + 1} de ${Estado.quizData.questions.length}`;
  }
  if (DOM.progressFill) {
    DOM.progressFill.style.width =
      `${(Estado.currentIndex / Estado.quizData.questions.length) * 100}%`;
  }

  iniciarTimer();
}

function selectOption(index, btn) {
  if (Estado.answered) return;

  btn.classList.add('rippling');
  setTimeout(() => btn.classList.remove('rippling'), 600);

  document.querySelectorAll('.option').forEach(o => o.classList.remove('selected'));
  btn.classList.add('selected');
  Estado.selectedIndex = index;

  somClique();

  const textoAlternativa = Estado.quizData.questions[Estado.currentIndex].options[index];
  falar(textoAlternativa);

  if (DOM.confirmBtn) {
    DOM.confirmBtn.disabled = false;
    DOM.confirmBtn.style.display = 'block';
  }
}

function confirmarResposta() {
  const btnConfirm = document.getElementById('confirmBtn');
  const btnNext = document.getElementById('nextBtn');

  if (!btnConfirm || !btnNext) {
    console.error('[ERRO CRÍTICO] Botões não encontrados no DOM!');
    return;
  }

  if (Estado.answered || Estado.selectedIndex === null || !Estado.quizData) return;

  Estado.answered = true;
  pararTimer();
  speechSynthesis.cancel();

  const q = Estado.quizData.questions[Estado.currentIndex];
  const allOptions = document.querySelectorAll('.option');
  const btnClicado = allOptions[Estado.selectedIndex];

  let cx = window.innerWidth / 2;
  let cy = window.innerHeight / 2;
  if (btnClicado) {
    const rect = btnClicado.getBoundingClientRect();
    cx = rect.left + rect.width / 2;
    cy = rect.top + rect.height / 2;
  }

  allOptions.forEach(o => {
    o.classList.add('disabled');
    o.classList.remove('selected');
  });

  const acertou = Estado.selectedIndex === q.correct;

  if (acertou) {
    allOptions[q.correct].classList.add('correct');
    allOptions[q.correct].querySelector('span')
      .insertAdjacentHTML('afterend', '<span class="icon">✔</span>');

    Estado.combo++;
    Estado.score++;

    if (DOM.scoreBadge) {
      DOM.scoreBadge.textContent = `⭐ ${Estado.score} pts`;
      DOM.scoreBadge.classList.add('bump');
      setTimeout(() => DOM.scoreBadge.classList.remove('bump'), 700);
    }

    somAcerto();
    screenFlash('green');
    celebraContainer();
    explodirParticulas(cx, cy, 35);
    burstConfetti(40);
    mostrarPontosFlutuantes(cx, cy - 30, '+1', '#38ef7d');

    if (Estado.combo >= 2) {
      setTimeout(() => {
        mostrarCombo(Estado.combo);
        somCombo(Estado.combo);
        if (Estado.combo >= 3) burstConfetti(30);
      }, 300);
    }
  } else {
    Estado.combo = 0;

    allOptions[q.correct].classList.add('correct');
    allOptions[q.correct].querySelector('span')
      .insertAdjacentHTML('afterend', '<span class="icon">✔</span>');

    allOptions[Estado.selectedIndex].classList.add('wrong');
    allOptions[Estado.selectedIndex].querySelector('span')
      .insertAdjacentHTML('afterend', '<span class="icon">✖</span>');

    somErro();
    screenFlash('red');
    screenShake();
    tremeContainer();
    explodirParticulas(cx, cy, 15, ['#96100E', '#d4181a', '#D66C25']);
  }

  const msg = acertou
    ? 'Resposta correta! Parabéns!'
    : `Resposta incorreta. A resposta correta é: ${q.options[q.correct]}`;
  setTimeout(() => falar(msg), acertou ? 500 : 400);

  if (DOM.progressFill) {
    DOM.progressFill.style.width =
      `${((Estado.currentIndex + 1) / Estado.quizData.questions.length) * 100}%`;
  }

  // Esconde confirmar
  btnConfirm.style.display = 'none';
  btnConfirm.disabled = true;

  // Mostra próxima
  btnNext.textContent = Estado.currentIndex === Estado.quizData.questions.length - 1
    ? 'Ver Resultado 🏁'
    : 'Próxima ➜';
  btnNext.disabled = false;
  btnNext.style.display = 'block';
  btnNext.style.opacity = '1';
  btnNext.style.pointerEvents = 'auto';
  btnNext.style.transform = 'translateY(0)';
}

function irParaProxima() {
  console.log('[DEBUG] irParaProxima. currentIndex antes:', Estado.currentIndex);

  speechSynthesis.cancel();
  pararTimer();

  Estado.currentIndex++;
  console.log('[DEBUG] currentIndex depois:', Estado.currentIndex);

  if (Estado.currentIndex < Estado.quizData.questions.length) {
    loadQuestion();
  } else {
    showResult();
  }
}

function showResult() {
  pararTimer();
  if (DOM.quizContent) DOM.quizContent.classList.add('hidden');
  if (DOM.resultScreen) DOM.resultScreen.classList.remove('hidden');

  const total = Estado.quizData.questions.length;
  const percent = (Estado.score / total) * 100;

  if (DOM.finalScore) DOM.finalScore.textContent = `${Estado.score} / ${total}`;

  const emoji = DOM.resultEmoji;
  const msg = DOM.resultMessage;

  if (percent === 100) {
    if (emoji) emoji.textContent = '🏆';
    if (msg) msg.textContent = 'PERFEITO! Você é um MESTRE! 🌟';
    burstConfetti(200);
    setTimeout(() => burstConfetti(150), 400);
    setTimeout(() => burstConfetti(100), 800);
    somVitoria();
    screenFlash('green');
    for (let i = 0; i < 5; i++) {
      setTimeout(() => {
        explodirParticulas(
          window.innerWidth * Math.random(),
          window.innerHeight * Math.random(),
          25
        );
      }, i * 200);
    }
  } else if (percent >= 70) {
    if (emoji) emoji.textContent = '🎉';
    if (msg) msg.textContent = 'Muito bem! Excelente desempenho!';
    burstConfetti(100);
    setTimeout(() => burstConfetti(80), 400);
    somVitoria();
  } else if (percent >= 40) {
    if (emoji) emoji.textContent = '👍';
    if (msg) msg.textContent = 'Bom trabalho! Continue praticando.';
    somAcerto();
  } else {
    if (emoji) emoji.textContent = '📚';
    if (msg) msg.textContent = 'Continue estudando, você vai melhorar!';
    somDerrota();
  }

  if (DOM.progressFill) DOM.progressFill.style.width = '100%';
  if (DOM.progressInfo) DOM.progressInfo.textContent = 'Concluído!';
}

function reiniciarQuiz() {
  speechSynthesis.cancel();
  pararTimer();

  Estado.currentIndex = 0;
  Estado.score = 0;

  if (DOM.progressFill) DOM.progressFill.style.width = '0%';
  if (DOM.resultScreen) DOM.resultScreen.classList.add('hidden');
  if (DOM.quizContent) DOM.quizContent.classList.remove('hidden');

  somClique();
  loadQuestion();
}

  if (DOM.soundBtn) {
    DOM.soundBtn.addEventListener('click', () => {
      Estado.somAtivado = !Estado.somAtivado;
      salvarSom();
      atualizarBotaoSom();

      if (Estado.somAtivado) {
        somClique();
        mostrarToast('🔊 Som ligado');
      } else {
        speechSynthesis.cancel();
        mostrarToast('🔇 Som desligado');
      }
    });
  }

  // Botão "ouvir pergunta"
  if (DOM.speakBtn) {
    DOM.speakBtn.addEventListener('click', () => {
      if (!('speechSynthesis' in window)) return;
      if (!Estado.somAtivado) {
        mostrarToast('🔇 Ative o som para ouvir');
        return;
      }
      speechSynthesis.cancel();

      if (DOM.speakBtn.classList.contains('speaking')) {
        DOM.speakBtn.classList.remove('speaking');
        return;
      }

      const q = Estado.quizData.questions[Estado.currentIndex];
      let texto = `Pergunta ${Estado.currentIndex + 1}. ${q.question}. As alternativas são: `;
      q.options.forEach((opt, i) => { texto += `Alternativa ${i + 1}: ${opt}. `; });

      const utter = falar(texto, { onend: () => DOM.speakBtn.classList.remove('speaking') });
      if (utter) {
        utter.onstart = () => DOM.speakBtn.classList.add('speaking');
        utter.onerror = () => DOM.speakBtn.classList.remove('speaking');
      }
    });
  }
}

// ============ TIMER REGRESSIVO ============

function iniciarTimer() {
  pararTimer();

  if (!configAtiva.timerAtivo) {
    if (DOM.timerBar) DOM.timerBar.style.display = 'none';
    if (DOM.timerLabel) DOM.timerLabel.style.display = 'none';
    return;
  }
  if (DOM.timerBar) DOM.timerBar.style.display = 'block';
  if (DOM.timerLabel) DOM.timerLabel.style.display = 'block';

  Estado.timeLeft = configAtiva.tempoPergunta;
  atualizarTimerUI();

  Estado.timerInterval = setInterval(() => {
    Estado.timeLeft -= 0.1;
    if (Estado.timeLeft <= 0) {
      Estado.timeLeft = 0;
      atualizarTimerUI();
      pararTimer();
      tempoEsgotado();
    } else {
      atualizarTimerUI();
      if (Estado.timeLeft <= 5 && Math.abs((Estado.timeLeft * 10) % 10) < 0.5) {
        somTique();
      }
    }
  }, 100);
}

function pararTimer() {
  if (Estado.timerInterval) {
    clearInterval(Estado.timerInterval);
    Estado.timerInterval = null;
  }
}

function atualizarTimerUI() {
  if (!configAtiva.timerAtivo) return;
  if (!DOM.timerFill || !DOM.timerLabel) return;

  const pct = (Estado.timeLeft / configAtiva.tempoPergunta) * 100;
  DOM.timerFill.style.width = pct + '%';
  DOM.timerLabel.textContent = `⏱️ ${Math.ceil(Estado.timeLeft)}s`;

  DOM.timerFill.classList.remove('warning', 'danger');
  DOM.timerLabel.classList.remove('warning', 'danger');

  if (pct <= 20) {
    DOM.timerFill.classList.add('danger');
    DOM.timerLabel.classList.add('danger');
  } else if (pct <= 50) {
    DOM.timerFill.classList.add('warning');
    DOM.timerLabel.classList.add('warning');
  }
}

// Só lista locutores em português (pt-BR, pt-PT, pt-*)

function listarVozes() {
  const todasVozes = speechSynthesis.getVoices();
  if (todasVozes.length === 0) return;

  // Filtra APENAS português
  const vozesPT = todasVozes.filter(v => v.lang.toLowerCase().startsWith('pt'));

  // Se não achou nenhuma voz PT, avisa e mantém vazio
  if (vozesPT.length === 0) {
    console.warn('[voz] Nenhuma voz em português encontrada neste navegador.');
    Estado.vozesDisponiveis = [];

    if (DOM.voiceSelect) {
      DOM.voiceSelect.innerHTML = '';
      const opt = document.createElement('option');
      opt.textContent = 'Nenhuma voz em português disponível';
      opt.disabled = true;
      opt.selected = true;
      DOM.voiceSelect.appendChild(opt);
    }
    return;
  }

  // Ordena: pt-BR primeiro, depois outros pt
  const ptBR = vozesPT.filter(v => v.lang.toLowerCase() === 'pt-br');
  const ptOutros = vozesPT.filter(v => v.lang.toLowerCase() !== 'pt-br');
  Estado.vozesDisponiveis = [...ptBR, ...ptOutros];

  if (!DOM.voiceSelect) return;

  DOM.voiceSelect.innerHTML = '';
  Estado.vozesDisponiveis.forEach((voz, i) => {
    const opt = document.createElement('option');
    opt.value = i;
    opt.textContent = `${voz.name} — ${voz.lang}`;
    DOM.voiceSelect.appendChild(opt);
  });

  const modalAberto = DOM.settingsModal && DOM.settingsModal.classList.contains('show');
  if (modalAberto) {
    selecionarVozNoSelect(configTemp.voiceName, configTemp.voiceLang);
  } else {
    selecionarVozNoSelect(configAtiva.voiceName, configAtiva.voiceLang);
  }
}

function selecionarVozNoSelect(name, lang) {
  if (!DOM.voiceSelect || Estado.vozesDisponiveis.length === 0) return;

  if (!name) {
    // Sem voz salva — pega a primeira pt-BR, ou a primeira da lista
    const idx = Estado.vozesDisponiveis.findIndex(v => v.lang.toLowerCase() === 'pt-br');
    DOM.voiceSelect.selectedIndex = idx >= 0 ? idx : 0;
    return;
  }

  let idx = Estado.vozesDisponiveis.findIndex(v => v.name === name && v.lang === lang);
  if (idx === -1) idx = Estado.vozesDisponiveis.findIndex(v => v.name === name);
  if (idx === -1) idx = Estado.vozesDisponiveis.findIndex(v => v.lang.toLowerCase() === 'pt-br');
  DOM.voiceSelect.selectedIndex = idx >= 0 ? idx : 0;
}

function obterVozAtiva() {
  if (Estado.vozesDisponiveis.length === 0) return null;

  if (!configAtiva.voiceName) {
    return Estado.vozesDisponiveis.find(v => v.lang.toLowerCase() === 'pt-br')
        || Estado.vozesDisponiveis[0];
  }
  return Estado.vozesDisponiveis.find(v => v.name === configAtiva.voiceName && v.lang === configAtiva.voiceLang)
      || Estado.vozesDisponiveis.find(v => v.name === configAtiva.voiceName)
      || Estado.vozesDisponiveis.find(v => v.lang.toLowerCase() === 'pt-br')
      || Estado.vozesDisponiveis[0];
}

function obterVozTemp() {
  if (!DOM.voiceSelect || Estado.vozesDisponiveis.length === 0) return null;
  return Estado.vozesDisponiveis[DOM.voiceSelect.selectedIndex] || null;
}

function falar(texto, opts = {}) {
  if (!('speechSynthesis' in window)) return null;
  if (!Estado.somAtivado) return null;

  const cfg = opts.useTemp ? configTemp : configAtiva;
  const voz = opts.useTemp ? obterVozTemp() : obterVozAtiva();

  speechSynthesis.cancel();

  const utter = new SpeechSynthesisUtterance(texto);
  if (voz) utter.voice = voz;
  utter.lang = voz ? voz.lang : 'pt-BR';
  utter.rate = cfg.rate;
  utter.pitch = cfg.pitch;
  utter.volume = cfg.volume;
  if (opts.onend) utter.onend = opts.onend;

  speechSynthesis.speak(utter);
  return utter;
}

function getAudioCtx() {
  if (!audioCtx) {
    audioCtx = new (window.AudioContext || window.webkitAudioContext)();
  }
  if (audioCtx.state === 'suspended') audioCtx.resume();
  return audioCtx;
}

function tocarTom(freq, dur, type = 'sine', volume = 0.15, delay = 0) {
  if (!somAtivado) return;
  try {
    const ctx = getAudioCtx();
    const osc = ctx.createOscillator();
    const gain = ctx.createGain();
    osc.type = type;
    osc.frequency.value = freq;
    osc.connect(gain);
    gain.connect(ctx.destination);
    const start = ctx.currentTime + delay;
    gain.gain.setValueAtTime(0, start);
    gain.gain.linearRampToValueAtTime(volume, start + 0.01);
    gain.gain.exponentialRampToValueAtTime(0.001, start + dur);
    osc.start(start);
    osc.stop(start + dur);
  } catch (e) { console.warn('Erro áudio:', e); }
}

function listarVozes() {
  const vozes = speechSynthesis.getVoices();
  if (vozes.length === 0) return;

  const ptBR = vozes.filter(v => v.lang === 'pt-BR');
  const ptOutros = vozes.filter(v => v.lang.startsWith('pt') && v.lang !== 'pt-BR');
  const outras = vozes.filter(v => !v.lang.startsWith('pt'));
  vozesDisponiveis = [...ptBR, ...ptOutros, ...outras];

  if (voiceSelect) {
    voiceSelect.innerHTML = '';
    vozesDisponiveis.forEach((voz, i) => {
      const opt = document.createElement('option');
      opt.value = i;
      opt.textContent = `${voz.name} — ${voz.lang}`;
      voiceSelect.appendChild(opt);
    });

    if (settingsModal && settingsModal.classList.contains('show')) {
      selecionarVozNoSelect(configTemp.voiceName, configTemp.voiceLang);
    } else {
      selecionarVozNoSelect(configAtiva.voiceName, configAtiva.voiceLang);
    }
  }
}

function selecionarVozNoSelect(name, lang) {
  if (!voiceSelect) return;
  if (!name) {
    const idx = vozesDisponiveis.findIndex(v => v.lang === 'pt-BR');
    voiceSelect.selectedIndex = idx >= 0 ? idx : 0;
    return;
  }
  let idx = vozesDisponiveis.findIndex(v => v.name === name && v.lang === lang);
  if (idx === -1) idx = vozesDisponiveis.findIndex(v => v.name === name);
  if (idx === -1) idx = vozesDisponiveis.findIndex(v => v.lang === 'pt-BR');
  voiceSelect.selectedIndex = idx >= 0 ? idx : 0;
}

function obterVozAtiva() {
  if (!configAtiva.voiceName) return vozesDisponiveis.find(v => v.lang === 'pt-BR') || vozesDisponiveis[0] || null;
  return vozesDisponiveis.find(v => v.name === configAtiva.voiceName && v.lang === configAtiva.voiceLang)
      || vozesDisponiveis.find(v => v.name === configAtiva.voiceName)
      || vozesDisponiveis.find(v => v.lang === 'pt-BR')
      || vozesDisponiveis[0] || null;
}
function obterVozTemp() { return voiceSelect ? (vozesDisponiveis[voiceSelect.selectedIndex] || null) : null; }

function falar(texto, opts = {}) {
  if (!('speechSynthesis' in window)) return null;
  if (!somAtivado) return null;
  const cfg = opts.useTemp ? configTemp : configAtiva;
  const voz = opts.useTemp ? obterVozTemp() : obterVozAtiva();
  synth.cancel();
  const utter = new SpeechSynthesisUtterance(texto);
  if (voz) utter.voice = voz;
  utter.lang = voz ? voz.lang : 'pt-BR';
  utter.rate = cfg.rate;
  utter.pitch = cfg.pitch;
  utter.volume = cfg.volume;
  if (opts.onend) utter.onend = opts.onend;
  synth.speak(utter);
  return utter;
}

function explodirParticulas(x, y, qtd = 25, cores = null) {
  const palette = cores || ['#00498B', '#0677CC', '#00BDB9', '#FBAB30', '#96100E', '#D66C25'];
  for (let i = 0; i < qtd; i++) {
    const p = document.createElement('div');
    p.className = 'particle';
    const cor = palette[Math.floor(Math.random() * palette.length)];
    p.style.background = cor;
    p.style.boxShadow = `0 0 15px ${cor}, 0 0 30px ${cor}`;
    p.style.left = x + 'px';
    p.style.top = y + 'px';
    const angulo = Math.random() * Math.PI * 2;
    const dist = Math.random() * 220 + 80;
    p.style.setProperty('--tx', Math.cos(angulo) * dist + 'px');
    p.style.setProperty('--ty', Math.sin(angulo) * dist + 'px');
    const size = Math.random() * 10 + 6;
    p.style.width = size + 'px';
    p.style.height = size + 'px';
    p.style.animationDuration = (Math.random() * 0.5 + 0.7) + 's';
    document.body.appendChild(p);
    setTimeout(() => p.remove(), 1500);
  }
}

function burstConfetti(amount) {
  const colors = ['#00498B', '#0677CC', '#00BDB9', '#FBAB30', '#96100E', '#D66C25'];
  for (let i = 0; i < amount; i++) {
    const conf = document.createElement('div');
    conf.className = 'confetti';
    conf.style.left = Math.random() * 100 + 'vw';
    conf.style.background = colors[Math.floor(Math.random() * colors.length)];
    conf.style.animationDuration = (Math.random() * 2.5 + 2) + 's';
    conf.style.animationDelay = (Math.random() * 0.8) + 's';
    conf.style.animationName = 'fall';
    conf.style.animationTimingFunction = 'ease-in';
    conf.style.animationFillMode = 'forwards';
    conf.style.borderRadius = Math.random() > 0.5 ? '50%' : '2px';
    const size = Math.random() * 10 + 6;
    conf.style.width = conf.style.height = size + 'px';
    document.body.appendChild(conf);
    setTimeout(() => conf.remove(), 5000);
  }
}

// ============ TIMER ============
function iniciarTimer() {
  pararTimer();

  if (!configAtiva.timerAtivo) {
    if (timerBar) timerBar.style.display = 'none';
    if (timerLabel) timerLabel.style.display = 'none';
    return;
  }
  if (timerBar) timerBar.style.display = 'block';
  if (timerLabel) timerLabel.style.display = 'block';

  timeLeft = configAtiva.tempoPergunta;
  atualizarTimerUI();

  timerInterval = setInterval(() => {
    timeLeft -= 0.1;
    if (timeLeft <= 0) {
      timeLeft = 0;
      atualizarTimerUI();
      pararTimer();
      tempoEsgotado();
    } else {
      atualizarTimerUI();
      if (timeLeft <= 5 && Math.abs((timeLeft * 10) % 10) < 0.5) {
        somTique();
      }
    }
  }, 100);
}

function pararTimer() {
  if (timerInterval) {
    clearInterval(timerInterval);
    timerInterval = null;
  }
}

function atualizarTimerUI() {
  if (!configAtiva.timerAtivo) return;
  if (!timerFill || !timerLabel) return;
  const pct = (timeLeft / configAtiva.tempoPergunta) * 100;
  timerFill.style.width = pct + '%';
  timerLabel.textContent = `⏱️ ${Math.ceil(timeLeft)}s`;

  timerFill.classList.remove('warning', 'danger');
  timerLabel.classList.remove('warning', 'danger');

  if (pct <= 20) {
    timerFill.classList.add('danger');
    timerLabel.classList.add('danger');
  } else if (pct <= 50) {
    timerFill.classList.add('warning');
    timerLabel.classList.add('warning');
  }
}

function tempoEsgotado() {
  if (answered) return;
  answered = true;
  combo = 0;

  synth.cancel();
  somTempoEsgotado();
  flashTimeUp();
  screenFlash('red');
  screenShake();
  tremeContainer();

  const q = quizData.questions[currentIndex];
  const allOptions = document.querySelectorAll('.option');

  allOptions.forEach(o => {
    o.classList.add('disabled');
    o.classList.remove('selected');
  });

  allOptions[q.correct].classList.add('correct');
  allOptions[q.correct].querySelector('span').insertAdjacentHTML('afterend', '<span class="icon">✔</span>');

  setTimeout(() => {
    falar(`Tempo esgotado! A resposta correta é: ${q.options[q.correct]}`);
  }, 400);

  progressFill.style.width = `${((currentIndex + 1) / quizData.questions.length) * 100}%`;

  // Esconde confirmar
  const btnConfirm = document.getElementById('confirmBtn');
  if (btnConfirm) {
    btnConfirm.style.display = 'none';
    btnConfirm.disabled = true;
  }

  // Mostra próxima
  const btnNext = document.getElementById('nextBtn');
  if (btnNext) {
    btnNext.textContent = currentIndex === quizData.questions.length - 1 ? 'Ver Resultado 🏁' : 'Próxima ➜';
    btnNext.disabled = false;
    btnNext.style.display = 'block';
    btnNext.style.opacity = '1';
    btnNext.style.pointerEvents = 'auto';
    btnNext.style.transform = 'translateY(0)';
  }
}

function iniciais(nome) {
  return nome.trim().split(/\s+/).slice(0, 2)
    .map(p => p[0] ? p[0].toUpperCase() : '').join('');
}

function criarLinks(pessoa) {
  const links = [];
  if (pessoa.linkedin && pessoa.linkedin.trim() !== '') {
    links.push(`<a class="about-link linkedin" href="${pessoa.linkedin}" target="_blank" rel="noopener" title="LinkedIn de ${pessoa.nome}">in</a>`);
  }
  if (pessoa.lattes && pessoa.lattes.trim() !== '') {
    links.push(`<a class="about-link lattes" href="${pessoa.lattes}" target="_blank" rel="noopener" title="Lattes de ${pessoa.nome}">CV</a>`);
  }
  return links.length ? `<div class="about-links">${links.join('')}</div>` : '';
}

function criarPessoa(pessoa, isProfessor = false) {
  const avatarClass = isProfessor ? 'about-avatar professor' : 'about-avatar';
  return `
    <div class="about-pessoa">
      <div class="about-pessoa-info">
        <div class="${avatarClass}">${iniciais(pessoa.nome)}</div>
        <div class="about-pessoa-nome">${pessoa.nome}</div>
      </div>
      ${criarLinks(pessoa)}
    </div>
  `;
}

function renderizarSobre(data) {
  const professorHTML = data.professor ? `
    <div class="about-secao">
      <div class="about-secao-titulo">${data.professor.titulo || 'Professor'}</div>
      ${criarPessoa(data.professor, true)}
    </div>` : '';

  const alunosHTML = (data.alunos && data.alunos.length) ? `
    <div class="about-secao">
      <div class="about-secao-titulo">Alunos</div>
      ${data.alunos.map(a => criarPessoa(a, false)).join('')}
    </div>` : '';

  if (aboutBody) {
    aboutBody.innerHTML = `
      <div>
        <div class="about-titulo">${data.titulo || 'Sobre o Projeto'}</div>
        <div class="about-descricao">${data.descricao || ''}</div>
      </div>
      ${professorHTML}
      ${alunosHTML}
      ${data.rodape ? `<div class="about-rodape">${data.rodape}</div>` : ''}
    `;
  }
}

// ============ QUIZ ============
function loadQuestion() {
  console.log('[DEBUG] loadQuestion — currentIndex:', currentIndex);

  answered = false;
  selectedIndex = null;

  if (!quizData || !quizData.questions || !quizData.questions[currentIndex]) {
    console.error('[ERRO] quizData inválido');
    return;
  }

  const q = quizData.questions[currentIndex];

  if (questionText) {
    questionText.textContent = q.question;
    questionText.style.animation = 'none';
    void questionText.offsetWidth;
    questionText.style.animation = '';
  }

  if (optionsContainer) {
    optionsContainer.innerHTML = '';

    q.options.forEach((opt, i) => {
      const btn = document.createElement('button');
      btn.className = 'option';
      btn.dataset.index = i;
      const spanText = document.createElement('span');
      spanText.textContent = opt;
      btn.appendChild(spanText);
      btn.addEventListener('click', () => selectOption(i, btn));
      optionsContainer.appendChild(btn);
    });
  }

  // ESCONDE os botões — busca SEMPRE pelo ID
  const btnConfirm = document.getElementById('confirmBtn');
  const btnNext = document.getElementById('nextBtn');

  if (btnConfirm) {
    btnConfirm.style.display = 'none';
    btnConfirm.disabled = true;
  }
  if (btnNext) {
    btnNext.style.display = 'none';
  }

  if (progressInfo) progressInfo.textContent = `Pergunta ${currentIndex + 1} de ${quizData.questions.length}`;
  if (progressFill) progressFill.style.width = `${(currentIndex / quizData.questions.length) * 100}%`;

  iniciarTimer();
}

function selectOption(index, btn) {
  if (answered) return;

  btn.classList.add('rippling');
  setTimeout(() => btn.classList.remove('rippling'), 600);

  document.querySelectorAll('.option').forEach(o => o.classList.remove('selected'));
  btn.classList.add('selected');
  selectedIndex = index;

  somClique();

  const textoAlternativa = quizData.questions[currentIndex].options[index];
  falar(textoAlternativa);

  // MOSTRA o botão confirmar — busca pelo ID
  const btnConfirm = document.getElementById('confirmBtn');
  if (btnConfirm) {
    btnConfirm.disabled = false;
    btnConfirm.style.display = 'block';
  }
}

// ============ CONFIRMAR ============
function confirmarResposta() {
  // BUSCA os botões a cada clique — nunca guarda referência stale
  const btnConfirm = document.getElementById('confirmBtn');
  const btnNext = document.getElementById('nextBtn');

  console.log('[DEBUG] confirmarResposta. btnConfirm:', btnConfirm ? 'OK' : 'NULL', '| btnNext:', btnNext ? 'OK' : 'NULL');

  if (!btnConfirm || !btnNext) {
    console.error('[ERRO CRÍTICO] Botões não encontrados no DOM!');
    return;
  }

  if (answered || selectedIndex === null || !quizData) return;
  answered = true;
  pararTimer();
  synth.cancel();

  const q = quizData.questions[currentIndex];
  const allOptions = document.querySelectorAll('.option');
  const btnClicado = allOptions[selectedIndex];

  let cx = window.innerWidth / 2;
  let cy = window.innerHeight / 2;
  if (btnClicado) {
    const rect = btnClicado.getBoundingClientRect();
    cx = rect.left + rect.width / 2;
    cy = rect.top + rect.height / 2;
  }

  allOptions.forEach(o => {
    o.classList.add('disabled');
    o.classList.remove('selected');
  });

  const acertou = selectedIndex === q.correct;

  if (acertou) {
    allOptions[q.correct].classList.add('correct');
    allOptions[q.correct].querySelector('span').insertAdjacentHTML('afterend', '<span class="icon">✔</span>');

    combo++;
    score++;
    if (scoreBadge) {
      scoreBadge.textContent = `⭐ ${score} pts`;
      scoreBadge.classList.add('bump');
      setTimeout(() => scoreBadge.classList.remove('bump'), 700);
    }

    somAcerto();
    screenFlash('green');
    celebraContainer();
    explodirParticulas(cx, cy, 35);
    burstConfetti(40);
    mostrarPontosFlutuantes(cx, cy - 30, '+1', '#38ef7d');

    if (combo >= 2) {
      setTimeout(() => {
        mostrarCombo(combo);
        somCombo(combo);
        if (combo >= 3) burstConfetti(30);
      }, 300);
    }
  } else {
    combo = 0;
    allOptions[q.correct].classList.add('correct');
    allOptions[q.correct].querySelector('span').insertAdjacentHTML('afterend', '<span class="icon">✔</span>');
    allOptions[selectedIndex].classList.add('wrong');
    allOptions[selectedIndex].querySelector('span').insertAdjacentHTML('afterend', '<span class="icon">✖</span>');

    somErro();
    screenFlash('red');
    screenShake();
    tremeContainer();
    explodirParticulas(cx, cy, 15, ['#96100E', '#d4181a', '#D66C25']);
  }

  const msg = acertou
    ? 'Resposta correta! Parabéns!'
    : `Resposta incorreta. A resposta correta é: ${q.options[q.correct]}`;
  setTimeout(() => falar(msg), acertou ? 500 : 400);

  if (progressFill) progressFill.style.width = `${((currentIndex + 1) / quizData.questions.length) * 100}%`;

  // Esconde confirmar
  btnConfirm.style.display = 'none';
  btnConfirm.disabled = true;

  // MOSTRA próxima
  btnNext.textContent = currentIndex === quizData.questions.length - 1 ? 'Ver Resultado 🏁' : 'Próxima ➜';
  btnNext.disabled = false;
  btnNext.style.display = 'block';
  btnNext.style.opacity = '1';
  btnNext.style.pointerEvents = 'auto';
  btnNext.style.transform = 'translateY(0)';

  console.log('[DEBUG] Botão próxima exibido com sucesso');
}

if (confirmBtn) confirmBtn.addEventListener('click', confirmarResposta);

function showResult() {
  pararTimer();
  if (quizContent) quizContent.classList.add('hidden');
  if (resultScreen) resultScreen.classList.remove('hidden');

  const total = quizData.questions.length;
  const percent = (score / total) * 100;

  const finalScoreEl = document.getElementById('finalScore');
  if (finalScoreEl) finalScoreEl.textContent = `${score} / ${total}`;

  const emoji = document.getElementById('resultEmoji');
  const msg = document.getElementById('resultMessage');

  if (percent === 100) {
    if (emoji) emoji.textContent = '🏆';
    if (msg) msg.textContent = 'PERFEITO! Você é um MESTRE! 🌟';
    burstConfetti(200);
    setTimeout(() => burstConfetti(150), 400);
    setTimeout(() => burstConfetti(100), 800);
    somVitoria();
    screenFlash('green');
    for (let i = 0; i < 5; i++) {
      setTimeout(() => {
        explodirParticulas(
          window.innerWidth * Math.random(),
          window.innerHeight * Math.random(),
          25
        );
      }, i * 200);
    }
  } else if (percent >= 70) {
    if (emoji) emoji.textContent = '🎉';
    if (msg) msg.textContent = 'Muito bem! Excelente desempenho!';
    burstConfetti(100);
    setTimeout(() => burstConfetti(80), 400);
    somVitoria();
  } else if (percent >= 40) {
    if (emoji) emoji.textContent = '👍';
    if (msg) msg.textContent = 'Bom trabalho! Continue praticando.';
    somAcerto();
  } else {
    if (emoji) emoji.textContent = '📚';
    if (msg) msg.textContent = 'Continue estudando, você vai melhorar!';
    somDerrota();
  }

  if (progressFill) progressFill.style.width = '100%';
  if (progressInfo) progressInfo.textContent = 'Concluído!';
}

    utter.onerror = () => speakBtn.classList.remove('speaking');
  }
